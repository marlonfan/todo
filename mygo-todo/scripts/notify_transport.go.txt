//go:build native_notify_fixture

// Temporary Go overlay only. Real plugins/validation/serialization run, while
// the explicitly allowed test host is routed to a disposable loopback server.
package notify

import (
 "fmt"
 "net/http"
 "net/url"
 "os"
 "time"
)

type nativeNotifyTransport struct { target *url.URL; host string }
func (t nativeNotifyTransport) RoundTrip(r *http.Request) (*http.Response,error) {
 if r.URL.Hostname()!=t.host { return nil,fmt.Errorf("fixture rejects unexpected notification host") }
 clone:=r.Clone(r.Context())
 u:=*r.URL
 u.Scheme,u.Host=t.target.Scheme,t.target.Host
 clone.URL,clone.Host=&u,t.target.Host
 transport:=http.DefaultTransport.(*http.Transport).Clone()
 transport.Proxy=nil
 defer transport.CloseIdleConnections()
 return transport.RoundTrip(clone)
}
func NativeFixtureHTTPClient(host string) *http.Client {
 target,err:=url.Parse(os.Getenv("TODO_NATIVE_NOTIFY_FIXTURE_URL"))
 if err!=nil || target==nil || target.Scheme!="http" || target.Hostname()!="127.0.0.1" || target.Port()=="" { panic("fixture requires an explicit local notification server") }
 if host!="notify.example.test" && host!="api.telegram.org" { panic("fixture requires an explicit synthetic notification host") }
 return &http.Client{Timeout:5*time.Second,Transport:nativeNotifyTransport{target,host},CheckRedirect:func(*http.Request,[]*http.Request) error { return http.ErrUseLastResponse }}
}
